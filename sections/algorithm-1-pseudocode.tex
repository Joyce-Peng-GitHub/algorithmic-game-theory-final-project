\section{The Full-Information Algorithm}

Algorithm~\ref{alg:full-information} reproduces the main algorithm of Deligkas, Fasoulakis, and Markakis \cite[Algorithm~1]{12WSNE}. Its detailed derivation and correctness argument appear in Appendix~\ref{app:algorithm-one-explanation}.

\begin{algorithm}[H]
  \caption{Main Algorithm for \((1/2+\delta)\)-WSNE}
  \label{alg:full-information}
  \small
  \textbf{Input:} A bimatrix game \((\mathcal{R},\mathcal{C})\), constants
  \(\delta\in(0,1]\), and
  \(\kappa(\delta)=\left\lceil 2\ln(1/\delta)/\delta^2\right\rceil\).\par
  \textbf{Output:} A \((1/2+\delta)\)-WSNE for \((\mathcal{R},\mathcal{C})\).
  \begin{enumerate}[label=\arabic*.,leftmargin=1.8em,itemsep=0.25em,topsep=0.4em]
    \item Compute a Nash equilibrium \((\mathbf{x}^*,\mathbf{y}^*)\) for the zero-sum game \((\mathcal{R},-\mathcal{R})\).
    \item Compute a Nash equilibrium \((\hat{\mathbf{x}},\hat{\mathbf{y}})\) for the zero-sum game \((-\mathcal{C},\mathcal{C})\).
    \item If \(\mathcal{R}(\mathbf{x}^*,\mathbf{y}^*)\geq \mathcal{C}(\hat{\mathbf{x}},\hat{\mathbf{y}})\):
      \begin{enumerate}[label=(\alph*),leftmargin=2.2em,itemsep=0.15em,topsep=0.1em]
        \item If \(\mathcal{R}(\mathbf{x}^*,\mathbf{y}^*)\leq 1/2\), then return \((\hat{\mathbf{x}},\mathbf{y}^*)\).
        \item Else, if there exists a strategy \(\mathbf{x}'\) such that
        \(\supp(\mathbf{x}')\subseteq\supp(\mathbf{x}^*)\) and
        \(\mathcal{C}(\mathbf{x}',e_j)\leq 1/2\) for every \(j\in[n]\), then return \((\mathbf{x}',\mathbf{y}^*)\).
        \item Else, exhaustively search over all \(\kappa(\delta)\)-uniform strategy profiles and find a \((1/2+\delta)\)-WSNE.
      \end{enumerate}
    \item Else if \(\mathcal{C}(\hat{\mathbf{x}},\hat{\mathbf{y}})>\mathcal{R}(\mathbf{x}^*,\mathbf{y}^*)\) (symmetric to Step~3, with the players' roles reversed):
      \begin{enumerate}[label=(\alph*),leftmargin=2.2em,itemsep=0.15em,topsep=0.1em]
        \item If \(\mathcal{C}(\hat{\mathbf{x}},\hat{\mathbf{y}})\leq 1/2\), then return \((\hat{\mathbf{x}},\mathbf{y}^*)\).
        \item Else if there exists a strategy \(\mathbf{y}'\) such that
        \(\supp(\mathbf{y}')\subseteq\supp(\hat{\mathbf{y}})\) and
        \(\mathcal{R}(e_i,\mathbf{y}')\leq 1/2\) for every \(i\in[n]\), then return \((\hat{\mathbf{x}},\mathbf{y}')\).
        \item Else, exhaustively search over all \(\kappa(\delta)\)-uniform strategy profiles and find a \((1/2+\delta)\)-WSNE.
      \end{enumerate}
  \end{enumerate}
\end{algorithm}
